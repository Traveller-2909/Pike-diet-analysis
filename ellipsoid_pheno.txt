
model{
    #priors
    for (k in 1:ngroups){
    mu[k,1] ~ dnorm(0,0.001)
    mu[k,2] ~ dnorm(0,0.001)
    mu[k,3] ~ dnorm(0,0.001)
    prec[1:3,1:3,k] ~ dwish(S3,4)
    cov[1:3,1:3,k] <- inverse(prec[1:3,1:3,k])
    #likelihood
    for (i in 1:maxn){
    y[i,1:3,k] ~ dmnorm(mu[k,],prec[,,k])
    }#i
    }
    }
